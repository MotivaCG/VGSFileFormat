# The decoder samples, built against an installed VGSDecoder.
#
# This is the copy that ships inside the install, so it is also the shortest proof that
# the install is self-contained: nothing here refers to the format's sources.
#
#   cmake -S . -B build -DCMAKE_PREFIX_PATH=<the INSTALL directory>
#   cmake --build build --config Release
#
# When this file sits in the examples folder of an install, the package is found without
# CMAKE_PREFIX_PATH: the hint below is the install root, one level up. A CMAKE_PREFIX_PATH
# given on the command line is searched first, so the same file also builds against an
# install somewhere else.

cmake_minimum_required(VERSION 3.16)
project(VGSDecoderExamples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(VGSDecoder REQUIRED HINTS "${CMAKE_CURRENT_SOURCE_DIR}/..")

foreach(sample vgsinfo vgsplay vgsdump vgsexport vgspagecost)
  add_executable(${sample} ${sample}.cpp)
  target_link_libraries(${sample} PRIVATE VGS::Decoder)
endforeach()
